IMPLEMENTATION

{ Non-negative number to decimal text, no leading zeros.  These are NUMERALS, not    }
{ language, so building them in code does not break the no-literal-strings rule.    }
procedure NumToStr {n : longint; var s : Str255};
var divisor : longint;
    d       : integer;
    started : boolean;
begin
   if n < 0 then n := 0;
   s := '';
   divisor := 1000000000;
   started := false;
   while divisor > 0 do begin
      d := (n div divisor) mod 10;
      if (d <> 0) or started or (divisor = 1) then begin
         s := concat(s, ' ');
         s[length(s)] := chr(ord('0') + d);
         started := true
      end;
      divisor := divisor div 10
   end
end;

{ Seconds from the battery-backed clock: real time, however fast the CPU runs.  (Get_Time has  }
{ one-second resolution; the Window Manager's Time and event times tick on the CPU clock, so    }
{ they run faster when the Lisa is overclocked.)                                                }
function RealSecs : longint;
var err : integer;
    t   : Time_Rec;
begin
   Get_Time(err, t);
   RealSecs := ((ord4(t.day) * 24 + t.hour) * 60 + t.minute) * 60 + t.second
end;

{ Start measuring how many Window Manager clock units make a real second. }
procedure InitClockCal {};
begin
   unitsPerSec := 100;
   calTime0 := Time;
   calSecs0 := RealSecs;
   nextCalPoll := 0
end;

{ Cheap enough to call on every idle pass: it reads the real clock only twice a second, and recomputes }
{ the speed every ten real seconds.                                                                  }
procedure CalibrateClock {};
var secs : longint;
begin
   if Time < nextCalPoll then exit(CalibrateClock);
   nextCalPoll := Time + unitsPerSec div 2;
   secs := RealSecs;
   if secs < calSecs0 then begin               { the year rolled over: start again }
      calSecs0 := secs;
      calTime0 := Time
   end
   else if secs - calSecs0 >= 10 then begin
      unitsPerSec := (Time - calTime0) div (secs - calSecs0);
      if unitsPerSec < 20 then unitsPerSec := 20;
      if unitsPerSec > 2000 then unitsPerSec := 2000;
      calTime0 := Time;
      calSecs0 := secs
   end
end;

{ n hundredths of a REAL second, expressed in Window Manager clock units. }
function ScaleUnits {(n : longint) : longint};
begin
   ScaleUnits := (n * unitsPerSec) div 100
end;

{ Fetch a phrase from the phrase file; empty if the number is missing. }
procedure GetPhrase {n : integer; var s : Str255};
begin
   s := '';
   GetAlert(allAlerts, n, @s)
end;

procedure LoadCaptions;
var k : CardKind;
begin
   for k := ck25 to sfRoW do begin
      GetPhrase(phCardBase + ord(k), capText[k])
   end;
   GetPhrase(phComputer, lblComputer);
   GetPhrase(phYou, lblYou);
   GetPhrase(phMiles, lblMiles);
   GetPhrase(phMatch, lblMatch);
   GetPhrase(phNext, lblNext)
end;

{ The Filer protocol.  EVERY Filer command must be answered, or the whole desktop      }
{ freezes.  fcNone is the only one that needs the main program: it lends us a window.  }
procedure HandleFilerOp {var wantsOpen : boolean};
var fx  : FilerExt;
    err : integer;
begin
   wantsOpen := false;
   GetAddParms(err, theEvent, fx);
   case fx.theFlrOp of
      fcNone :
         wantsOpen := true;
      fcResume :
         TellFiler(err, docClosd, internalError, theEvent.who);
      fcClose, fcSuspend, fcShred :
         begin
            myWindow := nil;
            TellFiler(err, docClosd, allOK, theEvent.who)
         end;
      fcCopy, fcPut :
         TellFiler(err, docNotXfered, internalError, theEvent.who);
      fcDfClose :
         TellFiler(err, dfClosed, allOK, theEvent.who);
      fcTerminate :
         toldToDie := true;
      otherwise
   end
end;
