{ MILES-CTRL: what a mouse click or a tick of the clock DOES to the game.  Human = seat 0,   }
{ computer = seat 1.  No Lisa units: tested on the Mac, compiled unchanged on the Lisa.     }
{ The Lisa program only turns events into calls here and draws the result.                  }
unit MilesCtrl;

interface

uses (*$U MILES/LOGIC.OBJ *) MilesLogic,
     (*$U MILES/LAYOUT.OBJ *) MilesLayout;

const
  { Status codes 1..12 are the Err* codes of MilesLogic; these are the UI's own. }
  MsgNone         = 0;
  MsgYourDraw     = 20;
  MsgYourPlay     = 21;
  MsgComputerTurn = 22;
  MsgCpuCoup = 23;
  MsgHandOver     = 24;
  MsgCoupAsk      = 25;
  MsgAlreadyDrew  = 26;
  MsgCpuPlayed    = 27;                { the computer played a card on itself }
  MsgCpuAttack    = 28;                { the computer attacked you }
  MsgCpuDiscard   = 29;                { the computer threw a card away }

type
  UIEvent = (ueNone, ueSelect, ueDeselect, ueDrew, ueMoved, ueRejected, ueCoupPrompt, ueHandOver,
             ueNextHand);

  UIState = record
    sel        : integer;                 { selected hand card, 0 = none }
    msg        : integer;                 { a message to show until the next action, or MsgNone }
    handScored : boolean;                 { this hand's points are already in match[] }
    hands      : integer;                 { matches dealt since the last New Game }
    lastClkAt  : longint;                 { time of the previous click on a hand card }
    lastClkIdx : integer;                 { which hand card it was, 0 = none (breaks any pair) }
    lastCard   : CardKind;                { the card the computer last played or threw away }
    match      : array[0..1] of longint;  { running match totals }
  end;

procedure UIInit(var u: UIState);
procedure UINewHand(var u: UIState; var g: Game; seed: longint);
procedure UINewGame(var u: UIState; var g: Game; seed: longint);
procedure UINextHand(var u: UIState; var g: Game);
{ now = the event's time, dblDelay = the double-click interval: the same units (hundredths of a second). }
procedure UIClick(var u: UIState; var g: Game; x, y: integer; now, dblDelay: longint; var ev: UIEvent);
procedure UIAdvance(var u: UIState; var g: Game; var ev: UIEvent);
procedure UICoupAnswer(var u: UIState; var g: Game; useCoup: boolean; var ev: UIEvent);
function  NaturalSlot(k: CardKind): integer;
function  StatusMsg(var u: UIState; var g: Game): integer;

implementation

procedure UIInit(var u: UIState);
begin
  u.sel := 0;
  u.msg := MsgNone;
  u.handScored := false;
  u.hands := 0;
  u.lastClkAt := 0;
  u.lastClkIdx := 0;
  u.lastCard := ckNone;
  u.match[0] := 0;
  u.match[1] := 0
end;

procedure UINewHand(var u: UIState; var g: Game; seed: longint);
begin
  NewGame(g, seed);
  u.sel := 0;
  u.msg := MsgNone;
  u.handScored := false
end;

{ New Game: forget every score and deal a fresh match. }
procedure UINewGame(var u: UIState; var g: Game; seed: longint);
begin
  UIInit(u);
  NewGame(g, seed)
end;

{ Next Match: deal again, keeping the running scores.  The seed comes from the finished deal's     }
{ generator, so every match differs with no clock needed, and tests stay reproducible.            }
procedure UINextHand(var u: UIState; var g: Game);
begin
  u.hands := u.hands + 1;
  u.lastClkIdx := 0;
  u.sel := 0;
  u.msg := MsgNone;
  u.handScored := false;
  NewGame(g, (g.rng mod 1000000) * 31 + u.hands)
end;

{ Which pile a card naturally lands on; the screen highlights it. }
function NaturalSlot(k: CardKind): integer;
begin
  NaturalSlot := SlotBattle;
  if IsDistance(k) then NaturalSlot := SlotMiles;
  if k = rmEndLimit then NaturalSlot := SlotSpeed;
  if k = hzLimit then NaturalSlot := SlotSpeed;
  if IsSafety(k) then NaturalSlot := SlotSafety0 + SafetyIdx(k)
end;

{ When the hand has ended, score it exactly once and report ueHandOver. }
procedure Finish(var u: UIState; var g: Game; var ev: UIEvent);
begin
  if g.ph = phOver then begin
    if not u.handScored then begin
      u.match[0] := u.match[0] + HandPoints(g, 0);
      u.match[1] := u.match[1] + HandPoints(g, 1);
      u.handScored := true
    end;
    u.sel := 0;
    ev := ueHandOver
  end
end;

{ After ANY successful human play: clear the selection, let the computer answer a Coup at once,   }
{ and score a finished match.                                                                      }
procedure HumanPlayed(var u: UIState; var g: Game; var ev: UIEvent);
begin
  u.sel := 0;
  u.msg := MsgNone;
  ev := ueMoved;
  if g.ph = phCoup then begin           { the computer is the victim: it always answers }
    ResolveCoup(g, true);
    u.msg := MsgCpuCoup
  end;
  Finish(u, g, ev)
end;

{ Double-click: play hand card i where it belongs if it can be played at all: on you first, else on  }
{ the computer (hazards).  It NEVER discards, so a slip cannot throw a card away.  If it cannot be   }
{ played, the card stays selected and the reason is shown.                                          }
procedure AutoPlay(var u: UIState; var g: Game; i: integer; var ev: UIEvent);
var err, errSelf, errOpp: integer; k: CardKind;
begin
  k := g.pl[0].hand[i];
  errSelf := CheckPlay(g, i, tgSelf);
  errOpp := CheckPlay(g, i, tgOpponent);
  if errSelf = ErrOK then begin
    DoPlay(g, i, tgSelf, err);
    HumanPlayed(u, g, ev)
  end
  else if errOpp = ErrOK then begin
    DoPlay(g, i, tgOpponent, err);
    HumanPlayed(u, g, ev)
  end
  else begin
    u.sel := i;
    if IsHazard(k) then u.msg := errOpp else u.msg := errSelf;
    ev := ueRejected
  end
end;

procedure UIClick(var u: UIState; var g: Game; x, y: integer; now, dblDelay: longint; var ev: UIEvent);
var hit: HitInfo; err: integer; t: Target; wasIdx: integer; wasAt: longint;
begin
  ev := ueNone;
  wasIdx := u.lastClkIdx;
  wasAt := u.lastClkAt;
  u.lastClkIdx := 0;                    { only a click on a hand card (below) can start a pair }
  if g.ph = phOver then begin
    HitTest(x, y, 0, 0, hit);
    if hit.kind = hkNext then begin
      UINextHand(u, g);
      ev := ueNextHand
    end
  end
  else if (g.ph = phDraw) and (g.turn = 0) then begin
    HitTest(x, y, g.pl[0].handN, 0, hit);
    if hit.kind = hkDraw then begin
      DrawCard(g);
      u.msg := MsgNone;
      ev := ueDrew
    end
    else begin
      u.msg := ErrPhase;
      ev := ueRejected
    end
  end
  else if (g.ph = phPlay) and (g.turn = 0) then begin
    HitTest(x, y, g.pl[0].handN, u.sel, hit);
    if hit.kind = hkHand then begin
      u.msg := MsgNone;
      if (hit.index = wasIdx) and (now >= wasAt) and (now - wasAt < dblDelay) then
        AutoPlay(u, g, hit.index, ev)   { a double click: play the card }
      else begin
        u.lastClkIdx := hit.index;
        u.lastClkAt := now;
        if hit.index = u.sel then begin
          u.sel := 0;
          ev := ueDeselect
        end
        else begin
          u.sel := hit.index;
          ev := ueSelect
        end
      end
    end
    else if hit.kind = hkDraw then begin
      u.msg := MsgAlreadyDrew;
      ev := ueRejected
    end
    else if (hit.kind = hkOwn) or (hit.kind = hkOpp) or (hit.kind = hkDiscard) then begin
      if u.sel = 0 then begin
        u.msg := ErrNoCard;
        ev := ueRejected
      end
      else begin
        if hit.kind = hkOwn then t := tgSelf
        else if hit.kind = hkOpp then t := tgOpponent
        else t := tgDiscard;
        DoPlay(g, u.sel, t, err);
        if err = ErrOK then
          HumanPlayed(u, g, ev)
        else begin
          u.msg := err;
          ev := ueRejected
        end
      end
    end
  end
end;

{ One computer action.  The Lisa program calls this on a timer so moves are visible. }
procedure UIAdvance(var u: UIState; var g: Game; var ev: UIEvent);
var mv: PlayMove; err: integer; k: CardKind;
begin
  ev := ueNone;
  if g.ph = phCoup then begin
    if g.coupSeat = 0 then begin
      u.msg := MsgCoupAsk;
      ev := ueCoupPrompt
    end
    else begin
      ResolveCoup(g, true);
      u.msg := MsgCpuCoup;
      ev := ueMoved;
      Finish(u, g, ev)
    end
  end
  else if (g.ph = phDraw) and (g.turn = 1) then begin
    DrawCard(g);
    u.msg := MsgNone;
    ev := ueDrew
  end
  else if (g.ph = phPlay) and (g.turn = 1) then begin
    AIChoose(g, mv);
    k := g.pl[1].hand[mv.hand];
    DoPlay(g, mv.hand, mv.tgt, err);
    u.lastCard := k;                    { so the screen can say what just happened }
    if mv.tgt = tgDiscard then u.msg := MsgCpuDiscard
    else if mv.tgt = tgOpponent then u.msg := MsgCpuAttack
    else u.msg := MsgCpuPlayed;
    ev := ueMoved;
    if g.ph = phCoup then begin
      if g.coupSeat = 0 then begin
        u.msg := MsgCoupAsk;
        ev := ueCoupPrompt
      end
    end;
    Finish(u, g, ev)
  end
  else if g.ph = phOver then
    Finish(u, g, ev)
end;

procedure UICoupAnswer(var u: UIState; var g: Game; useCoup: boolean; var ev: UIEvent);
begin
  ev := ueMoved;
  ResolveCoup(g, useCoup);
  u.msg := MsgNone;
  Finish(u, g, ev)
end;

function StatusMsg(var u: UIState; var g: Game): integer;
begin
  if u.msg <> MsgNone then StatusMsg := u.msg
  else if g.ph = phOver then StatusMsg := MsgHandOver
  else if g.ph = phCoup then StatusMsg := MsgCoupAsk
  else if g.turn = 1 then StatusMsg := MsgComputerTurn
  else if g.ph = phDraw then StatusMsg := MsgYourDraw
  else StatusMsg := MsgYourPlay
end;

end.
