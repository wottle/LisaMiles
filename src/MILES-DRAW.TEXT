procedure DrawContents {window : WindowPtr};
var all : Rect;
    s   : Str255;
begin
   if window = nil then exit(DrawContents);
   SetPort(window);
   SetRect(all, 0, 0, ContentW, ContentH);
   ClipRect(all);
   EraseRect(all);
   PenNormal;
   FrameRect(all);
   TextMode(srcOr);
   TextFace([]);
   TextFont(fntBig);
   s := lblComputer;
   MoveTo(20, 40);
   DrawString(s);
   s := lblYou;
   MoveTo(20, 160);
   DrawString(s);
   TextFont(fntCaption);
   GetPhrase(phStatusBase + StatusMsg(ui, theGame), s);
   MoveTo(8, 291);
   DrawString(s)
end;

end.
