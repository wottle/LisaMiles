{ MILES-DRAW.TEXT -- all QuickDraw rendering for 1000 Miles.  Geometry comes from MilesLayout,    }
{ state from the global game and UI records, text from the phrase file.  Cards are typographic    }
{ tiles: distance = plain frame with a big numeral; hazard = BLACK card; remedy = double frame;   }
{ safety = heavy frame with a black band.                                                        }
{                                                                                                }
{ NO FLICKER: DrawContents repaints everything and is used only when the window is first shown or }
{ uncovered.  After every game action RefreshContents repaints ONLY the pieces whose data changed }
{ since the last paint (it remembers what is on screen), and each piece erases just itself.  The  }
{ window as a whole is never blanked.  The highlight on the target pile is an inverted ring drawn }
{ INSIDE the pile, so redrawing the pile removes it.                                              }
unit MilesDraw;

INTERFACE

uses (*$U UnitStd.obj *)        UnitStd,
     (*$U UnitHz.obj *)         UnitHz,
     (*$U Storage.obj *)        Storage,
     (*$U QuickDraw.obj *)      QuickDraw,
     (*$U Fontmgr.obj *)        FontMgr,
     (*$U SysCall.obj *)        SysCall,
     (*$U wm.events.obj *)      Events,
     (*$U wm.folders.obj *)     Folders,
     (*$U wm.menus.obj *)       Menus,
     (*$U AlertMgr.obj *)       AlertMgr,
     (*$U WMLstd.obj *)         WMLstd,
     (*$U WMlCrs.obj *)         WMLCrs,
     (*$U FilerComm.obj *)      FilerComm,
     (*$U MILES/LOGIC.OBJ *)    MilesLogic,
     (*$U MILES/ICONS.OBJ *)    MilesIcons,
     (*$U MILES/LAYOUT.OBJ *)   MilesLayout,
     (*$U MILES/CTRL.OBJ *)     MilesCtrl,
     (*$U MILES/GLOBALS.OBJ *)  MilesGlobals;

{ Forget what is on screen (call once at startup): the next refresh will repaint everything. }
procedure InitDraw;

{ Repaint the whole table: first show, or an update event after the window was uncovered. }
procedure DrawContents (window : WindowPtr);

{ Repaint only what changed since the last paint. }
procedure RefreshContents (window : WindowPtr);

IMPLEMENTATION

const
   statusLineY = 274;
   statusTextY = 291;

var
   miPix  : array [1..15, 0..71] of integer;   { the card pictures, as QuickDraw bitmaps }
   miWhole  : Rect;        { the whole content area: the clip restored after each card }
   shownG    : Game;        { the game state as last painted }
   shownU    : UIState;     { the UI state as last painted }
   haveShown : boolean;     { false until the first full paint }

{ Unpack every icon into its bitmap array. }
procedure FillIcons;
var i, j : integer;
    buf  : IconBuf;
begin
   for i := 1 to NumIcons do begin
      LoadIcon(i, buf);
      for j := 0 to IconWords - 1 do
         miPix[i, j] := buf[j]
   end
end;

procedure InitDraw {};
begin
   haveShown := false;
   SetRect(miWhole, 0, 0, ContentW, ContentH);
   FillIcons
end;

procedure BoxToRect (var b : Box; var r : Rect);
begin
   { SetRect takes left, top, right, bottom -- NOT the record's field order. }
   SetRect(r, b.left, b.top, b.right, b.bottom)
end;

procedure CenterString (var r : Rect; baseline : integer; var s : Str255);
begin
   MoveTo(r.left + ((r.right - r.left) - StringWidth(s)) div 2, baseline);
   DrawString(s)
end;

{ A card back: a gray tile with a frame. }
procedure ShowBack (var r : Rect);
begin
   PenNormal;
   FillRect(r, gray);
   FrameRect(r)
end;

{ An empty pile: a gray outline. }
procedure DrawEmptySlot (var r : Rect);
begin
   EraseRect(r);
   PenNormal;
   PenPat(gray);
   FrameRect(r);
   PenPat(black)
end;

{ The target highlight: invert the outer 3 pixels of the pile.  It shows on white and black alike, }
{ and redrawing the pile (which erases it first) takes it away again.                              }
procedure DrawRing (var r : Rect);
var inner : Rect;
begin
   inner := r;
   InsetRect(inner, 3, 3);
   InvertRect(r);
   InvertRect(inner)
end;

{ Copy an icon bitmap onto the window at (left, top). }
procedure ShowIcon (icon, left, top : integer);
var src : BitMap;
    dst : Rect;
begin
   if icon = 0 then exit(ShowIcon);
   src.baseAddr := @miPix[icon, 0];
   src.rowBytes := IconRowBytes;
   SetRect(src.bounds, 0, 0, IconW, IconRows(icon));
   SetRect(dst, left, top, left + IconW, top + IconRows(icon));
   CopyBits(src, thePort^.portBits, src.bounds, dst, srcCopy, nil)
end;

{ A card face.  Everything drawn for the card is clipped to its own rectangle, so nothing can  }
{ spill onto a neighbour or be left behind when the card is redrawn; a caption is drawn only    }
{ if it fits inside the card.                                                                   }
procedure ShowFace (k : CardKind; var r : Rect);
var s     : Str255;
    fi    : FontInfo;
    inner : Rect;
    band  : Rect;
    room  : integer;
begin
   PenNormal;
   EraseRect(r);
   ClipRect(r);
   TextMode(srcOr);
   TextFace([]);
   room := (r.right - r.left) - 6;               { the widest caption that still sits inside the frame }
   if IsDistance(k) then begin
      FrameRect(r);
      TextFont(fntBig);
      GetFontInfo(fi);
      NumToStr(DistValue(k), s);
      CenterString(r, r.top + 5 + fi.ascent, s);
      ShowIcon(IconOf(k), r.left + 9, r.bottom - 17)        { the road, under the number }
   end
   else begin
      ShowIcon(IconOf(k), r.left + 9, r.top + 4);
      TextFont(fntCaption);
      GetFontInfo(fi);
      FitCaption(capText[k], room, s);                      { the longest caption that fits }
      if IsSafety(k) then begin
         PenSize(2, 2);
         FrameRect(r);
         PenNormal;
         SetRect(band, r.left + 2, r.bottom - 15, r.right - 2, r.bottom - 2);
         PaintRect(band);
         if length(s) > 0 then begin
            TextMode(srcBic);                              { white text on the black band }
            CenterString(r, band.bottom - 2 - fi.descent, s);
            TextMode(srcOr)
         end
      end
      else begin
         FrameRect(r);
         if IsRemedy(k) then begin
            inner := r;
            InsetRect(inner, 3, 3);
            FrameRect(inner)
         end;
         if length(s) > 0 then
            CenterString(r, r.bottom - 5 - fi.descent, s);
         if IsHazard(k) then
            InvertRect(r)                                  { hazards are black cards }
      end
   end;
   ClipRect(miWhole)
end;

{ The miles-travelled box: a framed number. }
procedure DrawMilesBox (miles : integer; var r : Rect);
var s  : Str255;
    fi : FontInfo;
begin
   PenNormal;
   EraseRect(r);
   FrameRect(r);
   TextMode(srcOr);
   TextFace([]);
   TextFont(fntBig);
   GetFontInfo(fi);
   NumToStr(miles, s);
   CenterString(r, r.top + 6 + fi.ascent, s);
   TextFont(fntCaption);
   GetFontInfo(fi);
   s := lblMiles;
   CenterString(r, r.bottom - 6 - fi.descent, s)
end;

{ A player's header line: the name, a 1000-mile trip bar, and the mileage as text. }
procedure DrawHeader (seat : integer);
var hb       : Box;
    r, bar   : Rect;
    s, t     : Str255;
    w        : integer;
begin
   SlotBox(seat, SlotBattle, hb);
   SetRect(r, 0, hb.top - 18, ContentW, hb.top - 2);
   PenNormal;
   EraseRect(r);
   TextMode(srcOr);
   TextFace([bold]);
   TextFont(fntCaption);
   MoveTo(8, hb.top - 6);
   if seat = 0 then s := lblYou else s := lblComputer;
   DrawString(s);
   TextFace([]);
   SetRect(bar, 130, hb.top - 16, 430, hb.top - 7);
   FrameRect(bar);
   w := ord4(theGame.pl[seat].miles) * 298 div 1000;
   if w > 0 then begin
      SetRect(r, 131, bar.top + 1, 131 + w, bar.bottom - 1);
      PaintRect(r)
   end;
   NumToStr(theGame.pl[seat].miles, s);
   t := lblMiles;
   s := concat(s, ' / 1000 ', t);
   MoveTo(440, hb.top - 6);
   DrawString(s)
end;

{ One of a player's seven piles (and its target ring, if it is the target). }
procedure DrawSlot (seat, slot : integer);
var b : Box;
    r : Rect;
    k : CardKind;
    i : integer;
begin
   SlotBox(seat, slot, b);
   BoxToRect(b, r);
   if slot = SlotBattle then begin
      if theGame.pl[seat].battle <> ckNone then ShowFace(theGame.pl[seat].battle, r)
      else DrawEmptySlot(r)
   end
   else if slot = SlotSpeed then begin
      if theGame.pl[seat].limited then ShowFace(hzLimit, r)
      else DrawEmptySlot(r)
   end
   else if slot = SlotMiles then
      DrawMilesBox(theGame.pl[seat].miles, r)
   else begin
      k := sfAce;
      for i := 1 to slot - SlotSafety0 do
         k := succ(k);
      if theGame.pl[seat].safe[slot - SlotSafety0] then ShowFace(k, r)
      else DrawEmptySlot(r)
   end;
   if UITarget(ui, theGame, seat) = slot then DrawRing(r)
end;

{ The computer's hidden hand: card backs on the right of its row. }
procedure DrawBacks;
var b : Box;
    r : Rect;
    i : integer;
begin
   OppHandBox(1, b);
   SetRect(r, b.left - 4, b.top, ContentW, b.bottom);
   EraseRect(r);
   for i := 1 to theGame.pl[1].handN do begin
      OppHandBox(i, b);
      BoxToRect(b, r);
      ShowBack(r)
   end
end;

{ The draw pile: a small stack of backs with the count on top. }
procedure ShowPile;
var b    : Box;
    r, c : Rect;
    s    : Str255;
    fi   : FontInfo;
    j    : integer;
begin
   DrawPileBox(b);
   BoxToRect(b, r);
   EraseRect(r);
   for j := 2 downto 0 do begin
      SetRect(r, b.left + (2 - j) * 2, b.top + j * 2,
                 b.left + (2 - j) * 2 + CardW, b.top + j * 2 + CardH);
      ShowBack(r)
   end;
   SetRect(c, r.left + 12, r.top + 15, r.right - 12, r.bottom - 15);
   EraseRect(c);
   FrameRect(c);
   TextFont(fntCaption);
   TextMode(srcOr);
   TextFace([]);
   GetFontInfo(fi);
   NumToStr(RemainingDeck(theGame), s);
   CenterString(c, c.top + (c.bottom - c.top + fi.ascent - fi.descent) div 2, s)
end;

procedure DrawDiscardArea;
var b : Box;
    r : Rect;
begin
   DiscardBox(b);
   BoxToRect(b, r);
   if theGame.discardTop <> ckNone then ShowFace(theGame.discardTop, r)
   else DrawEmptySlot(r)
end;

{ The running scores. }
procedure DrawScores;
var b    : Box;
    r    : Rect;
    s, t : Str255;
begin
   DiscardBox(b);
   SetRect(r, 130, b.top, 326, b.bottom);
   PenNormal;
   EraseRect(r);
   TextFont(fntCaption);
   TextMode(srcOr);
   TextFace([bold]);
   s := lblMatch;
   MoveTo(136, b.top + 14);
   DrawString(s);
   TextFace([]);
   t := lblYou;
   NumToStr(ui.match[0], s);
   s := concat(t, ' ', s);
   MoveTo(136, b.top + 30);
   DrawString(s);
   t := lblComputer;
   NumToStr(ui.match[1], s);
   s := concat(t, ' ', s);
   MoveTo(136, b.top + 46);
   DrawString(s)
end;

{ The Next Match button while a match is over; its spot is blank otherwise. }
procedure DrawNextButton;
var b  : Box;
    r  : Rect;
    s  : Str255;
    fi : FontInfo;
begin
   NextBox(b);
   BoxToRect(b, r);
   PenNormal;
   EraseRect(r);
   if (theGame.ph = phOver) and (not ui.gameOver) then begin
      PenSize(2, 2);
      FrameRect(r);
      PenNormal;
      TextMode(srcOr);
      TextFont(fntCaption);
      TextFace([bold]);
      GetFontInfo(fi);
      s := lblNext;
      CenterString(r, r.top + ((r.bottom - r.top) + fi.ascent - fi.descent) div 2, s);
      TextFace([])
   end
end;

{ Your hand, raised card and all.  The whole strip is erased first (a lifted card used to be lower). }
procedure DrawHandStrip;
var i : integer;
    b : Box;
    r : Rect;
begin
   SetRect(r, 0, 204, ContentW, 264);
   PenNormal;
   EraseRect(r);
   for i := 1 to theGame.pl[0].handN do begin
      HandBox(i, i = ui.sel, b);
      BoxToRect(b, r);
      ShowFace(theGame.pl[0].hand[i], r)
   end
end;

{ The status text under the separator line; messages about the computer's move name its card. }
procedure DrawStatusText;
var s, t : Str255;
    code : integer;
    r    : Rect;
begin
   SetRect(r, 0, statusLineY + 1, ContentW, ContentH);
   PenNormal;
   EraseRect(r);
   code := StatusMsg(ui, theGame);
   GetPhrase(phStatusBase + code, s);
   if (code = MsgYourPlay) and (ui.sel >= 1) then      { a card is selected: say what it is and does }
      GetPhrase(phDescBase + ord(theGame.pl[0].hand[ui.sel]), s);
   if (code >= MsgCpuPlayed) and (code <= MsgCpuDiscard) then begin
      FitCaption(capText[ui.lastCard], 9999, t);
      s := concat(s, ' ', t)
   end;
   TextFont(fntCaption);
   TextMode(srcOr);
   TextFace([]);
   MoveTo(8, statusTextY);
   DrawString(s)
end;

procedure Remember;
begin
   shownG := theGame;
   shownU := ui;
   haveShown := true
end;

procedure DrawContents {window : WindowPtr};
var all  : Rect;
    seat : integer;
    slot : integer;
begin
   if window = nil then exit(DrawContents);
   SetPort(window);
   SetRect(all, 0, 0, ContentW, ContentH);
   ClipRect(all);
   PenNormal;
   EraseRect(all);
   for seat := 1 downto 0 do begin
      DrawHeader(seat);
      for slot := 0 to NumSlots - 1 do
         DrawSlot(seat, slot)
   end;
   DrawBacks;
   ShowPile;
   DrawDiscardArea;
   DrawScores;
   DrawNextButton;
   DrawHandStrip;
   MoveTo(0, statusLineY);
   LineTo(ContentW - 1, statusLineY);
   DrawStatusText;
   Remember
end;

procedure RefreshContents {window : WindowPtr};
var all         : Rect;
    seat, slot  : integer;
    i           : integer;
    tOld, tNew  : integer;
    changed     : boolean;
    handChanged : boolean;
    wasBtn, isBtn : boolean;
begin
   if window = nil then exit(RefreshContents);
   if not haveShown then begin
      DrawContents(window);
      exit(RefreshContents)
   end;
   SetPort(window);
   SetRect(all, 0, 0, ContentW, ContentH);
   ClipRect(all);

   for seat := 1 downto 0 do begin
      if shownG.pl[seat].miles <> theGame.pl[seat].miles then DrawHeader(seat);
      tOld := UITarget(shownU, shownG, seat);
      tNew := UITarget(ui, theGame, seat);
      for slot := 0 to NumSlots - 1 do begin
         { a pile is redrawn if its contents changed, or it gained or lost the target ring }
         changed := ((tOld = slot) and (tNew <> slot)) or ((tOld <> slot) and (tNew = slot));
         if slot = SlotBattle then begin
            if shownG.pl[seat].battle <> theGame.pl[seat].battle then changed := true
         end
         else if slot = SlotSpeed then begin
            if shownG.pl[seat].limited <> theGame.pl[seat].limited then changed := true
         end
         else if slot = SlotMiles then begin
            if shownG.pl[seat].miles <> theGame.pl[seat].miles then changed := true
         end
         else begin
            if shownG.pl[seat].safe[slot - SlotSafety0] <> theGame.pl[seat].safe[slot - SlotSafety0] then
               changed := true
         end;
         if changed then DrawSlot(seat, slot)
      end
   end;

   if shownG.pl[1].handN <> theGame.pl[1].handN then DrawBacks;
   if RemainingDeck(shownG) <> RemainingDeck(theGame) then ShowPile;
   if shownG.discardTop <> theGame.discardTop then DrawDiscardArea;
   if (shownU.match[0] <> ui.match[0]) or (shownU.match[1] <> ui.match[1]) then DrawScores;
   wasBtn := (shownG.ph = phOver) and (not shownU.gameOver);   { was / is Next Match showing? }
   isBtn := (theGame.ph = phOver) and (not ui.gameOver);
   if (wasBtn and (not isBtn)) or ((not wasBtn) and isBtn) then DrawNextButton;

   handChanged := (shownG.pl[0].handN <> theGame.pl[0].handN) or (shownU.sel <> ui.sel);
   for i := 1 to theGame.pl[0].handN do
      if shownG.pl[0].hand[i] <> theGame.pl[0].hand[i] then handChanged := true;
   if handChanged then DrawHandStrip;

   if (StatusMsg(shownU, shownG) <> StatusMsg(ui, theGame)) or (shownU.lastCard <> ui.lastCard) or
      (shownU.sel <> ui.sel) then
      DrawStatusText;

   Remember
end;

end.
